// npm i microsoft-cognitiveservices-speech-sdk multer
const sdk = require("microsoft-cognitiveservices-speech-sdk");

app.post("/api/v1/pronunciation", upload.single("audio"), (req, res) => {
  const { referenceText, language } = req.body;

  const speechConfig = sdk.SpeechConfig.fromSubscription(
    process.env.AZURE_SPEECH_KEY, process.env.AZURE_SPEECH_REGION);
  speechConfig.speechRecognitionLanguage = language || "en-US";

  const pa = new sdk.PronunciationAssessmentConfig(
    referenceText,
    sdk.PronunciationAssessmentGradingSystem.HundredMark,
    sdk.PronunciationAssessmentGranularity.Phoneme,
    true);                                   // miscue detection

  const audioConfig = sdk.AudioConfig.fromWavFileInput(req.file.buffer);
  const recognizer  = new sdk.SpeechRecognizer(speechConfig, audioConfig);
  pa.applyTo(recognizer);

  recognizer.recognizeOnceAsync(result => {
    recognizer.close();
    if (result.reason !== sdk.ResultReason.RecognizedSpeech)
      return res.json({ indeterminate: true, reason: result.reason });

    // per-phoneme detail lives in the raw JSON, not the typed result
    const raw = JSON.parse(result.json);
    const nb  = raw.NBest?.[0] ?? {};
    res.json({
      recognized: result.text,
      overall:      nb.PronunciationAssessment?.PronScore,
      accuracy:     nb.PronunciationAssessment?.AccuracyScore,
      fluency:      nb.PronunciationAssessment?.FluencyScore,
      completeness: nb.PronunciationAssessment?.CompletenessScore,
      words: (nb.Words || []).map(w => ({
        word:     w.Word,
        accuracy: w.PronunciationAssessment?.AccuracyScore,
        errorType:w.PronunciationAssessment?.ErrorType,
        phonemes:(w.Phonemes || []).map(p => ({
          phoneme:  p.Phoneme,
          accuracy: p.PronunciationAssessment?.AccuracyScore }))
      }))
    });
  }, err => { recognizer.close(); res.status(500).json({ error: String(err) }); });
});
